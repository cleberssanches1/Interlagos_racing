1. Introdução e Objetivos

Este documento define a especificação técnica completa do sistema de física veicular do Interlagos Racing. O objetivo é um modelo que equilibre fidelidade física com jogabilidade responsiva — um ponto entre simulação pura (iRacing, Gran Turismo) e arcade (Mario Kart).

Princípio de Design
Cada subsistema é fisicamente correto na sua base matemática, mas expõe coeficientes ajustáveis para que o designer possa "tunar" a sensação sem quebrar a consistência interna.

Subsistemas Implementados

  • Cinemática: integração de posição/velocidade/aceleração via Verlet semi-implícito
  • Motor e Câmbio: curva de torque polinomial, relações de marcha, diferencial
  • Rodas e Suspensão: modelo de mola-amortecedor (spring-damper) por roda
  • Pneus: modelo de Pacejka simplificado (Magic Formula) para força lateral e longitudinal
  • Aerodinâmica: drag proporcional a v², downforce e wash lateral
  • Frenagem: distribuição de frenagem, bloqueio de roda e ABS digital
  • Direção: esterçamento com Ackermann geometry e retorno de centro
  • Colisões: impulsos com coeficiente de restituição e fricção de contato

2. Arquitetura do Sistema de Física

O sistema é organizado em camadas. A camada mais interna trata matemática pura; a camada do veículo agrega todos os subsistemas; a camada de integração avança o estado no tempo a cada tick fixo.

3. Vetores e Matemática Fundamental

Toda física depende de operações vetoriais. Definimos Vec3 como a estrutura central, com operadores sobrecarregados e utilitários comuns:

struct Vec3 {
    float x, y, z;

    Vec3() : x(0), y(0), z(0) {}
    Vec3(float x, float y, float z) : x(x), y(y), z(z) {}

    Vec3 operator+(const Vec3& o) const { return {x+o.x, y+o.y, z+o.z}; }
    Vec3 operator-(const Vec3& o) const { return {x-o.x, y-o.y, z-o.z}; }
    Vec3 operator*(float s)         const { return {x*s, y*s, z*s}; }
    Vec3 operator/(float s)         const { return {x/s, y/s, z/s}; }
    Vec3& operator+=(const Vec3& o)  { x+=o.x; y+=o.y; z+=o.z; return *this; }

    float dot(const Vec3& o) const { return x*o.x + y*o.y + z*o.z; }
    Vec3  cross(const Vec3& o) const {
        return { y*o.z - z*o.y, z*o.x - x*o.z, x*o.y - y*o.x };
    }
    float lengthSq() const    { return x*x + y*y + z*z; }
    float length()   const    { return std::sqrt(lengthSq()); }
    Vec3  normalized() const  {
        float l = length();
        return l > 1e-6f ? (*this) / l : Vec3{};
    }
    Vec3 projectOnto(const Vec3& n) const { return n * dot(n); }
    Vec3 rejectFrom(const Vec3& n)  const { return *this - projectOnto(n); }

    static Vec3 lerp(const Vec3& a, const Vec3& b, float t) {
        return a + (b - a) * t;
    }
};

Quaternion para Rotação

Rotações do corpo do carro são representadas por quaternions para evitar gimbal lock:

struct Quat {
    float w, x, y, z;

    Quat() : w(1), x(0), y(0), z(0) {}
    Quat(float w, float x, float y, float z) : w(w), x(x), y(y), z(z) {}

    // Cria quaternion a partir de eixo+ângulo (ângulo em radianos)
    static Quat fromAxisAngle(const Vec3& axis, float angle) {
        float s = std::sin(angle * 0.5f);
        return { std::cos(angle * 0.5f), axis.x*s, axis.y*s, axis.z*s };
    }

    Quat operator*(const Quat& q) const {
        return {
            w*q.w - x*q.x - y*q.y - z*q.z,
            w*q.x + x*q.w + y*q.z - z*q.y,
            w*q.y - x*q.z + y*q.w + z*q.x,
            w*q.z + x*q.y - y*q.x + z*q.w
        };
    }

    Vec3 rotate(const Vec3& v) const {
        Vec3 u{x, y, z};
        return u * (2.0f * u.dot(v))
             + v * (w*w - u.dot(u))
             + u.cross(v) * (2.0f * w);
    }

    Quat normalized() const {
        float n = std::sqrt(w*w + x*x + y*y + z*z);
        return {w/n, x/n, y/n, z/n};
    }
};

4. Cinemática — Posição, Velocidade e Aceleração

A cinemática descreve o movimento de um objeto sem considerar suas causas. A equação fundamental é:

xf = xi + v·Δt + ½·a·Δt²
vf = vi + a·Δt
// Onde: x=posição, v=velocidade, a=aceleração, Δt=timestep

No sistema do carro, usamos integração semi-implícita de Euler (Symplectic Euler), que é mais estável que Euler explícito para sistemas oscilatórios como suspensões:

struct RigidBody3D {
    Vec3 position;      // metros, espaço mundial
    Vec3 velocity;      // m/s
    Vec3 acceleration;  // m/s²  (calculado em cada frame)

    Quat orientation;          // rotação atual
    Vec3 angularVelocity;      // rad/s em espaço local
    Vec3 angularAcceleration;  // rad/s²

    float mass;               // kg
    Vec3  inertiaTensor;      // kg·m² (diagonal simplificada)

    Vec3 forceAccumulator;    // reset a cada tick
    Vec3 torqueAccumulator;

    void addForce(const Vec3& f) { forceAccumulator += f; }
    void addForceAtPoint(const Vec3& f, const Vec3& worldPoint) {
        forceAccumulator += f;
        Vec3 r = worldPoint - position;
        torqueAccumulator += r.cross(f);
    }

    // Integração Symplectic Euler — atualiza velocidade antes da posição
    void integrate(float dt) {
        // Linear
        acceleration = forceAccumulator / mass;
        velocity    += acceleration * dt;        // v_new = v + a·dt
        position    += velocity * dt;            // x_new = x + v_new·dt

        // Angular
        angularAcceleration = {
            torqueAccumulator.x / inertiaTensor.x,
            torqueAccumulator.y / inertiaTensor.y,
            torqueAccumulator.z / inertiaTensor.z
        };
        angularVelocity += angularAcceleration * dt;

        // Integra orientação via quaternion
        Quat spin = Quat{
            0,
            angularVelocity.x * 0.5f * dt,
            angularVelocity.y * 0.5f * dt,
            angularVelocity.z * 0.5f * dt
        };
        orientation = (orientation * spin).normalized();

        // Limpa acumuladores para o próximo tick
        forceAccumulator  = {};
        torqueAccumulator = {};
    }

    // Converte direção local → global
    Vec3 localToWorld(const Vec3& local) const {
        return orientation.rotate(local);
    }

    Vec3 forward() const { return localToWorld({0, 0, 1}); }
    Vec3 right()   const { return localToWorld({1, 0, 0}); }
    Vec3 up()      const { return localToWorld({0, 1, 0}); }

    float speedKmh() const { return velocity.length() * 3.6f; }
};
Timestep Fixo
Sempre integre a física com um dt fixo (ex: 1/120 s). Nunca passe o tempo real do frame diretamente — acumule o tempo restante e chame integrate() em loop até consumir o acumulador.

5. Forças, Massa e Segunda Lei de Newton

A Segunda Lei de Newton (F = ma) é o pilar de toda a dinâmica. Reescrevendo: a = F/m. Quanto maior a massa, menor a aceleração para a mesma força.

Fnet = Fmotor + Ffreio + Fnormal + Fatrito + Faero + Fgravidade

Gravidade e Força Normal

O peso do carro gera a força normal em cada roda, que por sua vez determina o atrito disponível. Em terreno plano:

W = m · g = 1200 · 9.81 = 11772 N
Ndianteiro = W · (1 - cg_bias) // cg_bias=0.45 → 55% no eixo dianteiro
Ntraseiro = W · cg_bias

Em curvas, a transferência de carga lateral é crítica para o comportamento do carro:

ΔNlateral = m · alat · hCG / track_width
// Transferência de peso para o lado externo da curva
struct WheelLoad {
    float fl, fr, rl, rr;  // Newtons em cada roda
};

WheelLoad computeWheelLoads(const CarConfig& cfg,
                              const Vec3& accel_local,
                              float g = 9.81f)
{
    float W      = cfg.mass_kg * g;
    float a_long = accel_local.z;   // aceleração longitudinal (frente=+)
    float a_lat  = accel_local.x;   // aceleração lateral (direita=+)
    float L      = cfg.wheelbase_m;
    float tw     = cfg.track_width_m;
    float h      = cfg.cg_height_m;

    // Distribuição longitudinal estática
    float W_front = W * (1.0f - cfg.cg_bias);
    float W_rear  = W * cfg.cg_bias;

    // Transferência longitudinal por aceleração/frenagem
    float dN_long = cfg.mass_kg * a_long * h / L;
    W_front -= dN_long;
    W_rear  += dN_long;

    // Transferência lateral (mesma lógica para cada eixo)
    float dN_lat = cfg.mass_kg * a_lat * h / tw;

    WheelLoad out;
    out.fl = std::max(0.0f, W_front * 0.5f - dN_lat);
    out.fr = std::max(0.0f, W_front * 0.5f + dN_lat);
    out.rl = std::max(0.0f, W_rear  * 0.5f - dN_lat);
    out.rr = std::max(0.0f, W_rear  * 0.5f + dN_lat);
    return out;
}

6. Motor, Curva de Torque e Transmissão

O motor gera torque em função do RPM. A curva real é não-linear e é aproximada por uma função polinomial cúbica ou uma tabela com interpolação linear.

Troda = Tmotor · Gmarcha · Gfinal · ηdriveline
Ftração = Troda / rpneu
// G=relação de transmissão, η≈0.85 (perda por atrito interno)
class Engine {
public:
    struct Config {
        float idle_rpm      = 800.0f;
        float redline_rpm   = 7500.0f;
        float max_torque_nm = 350.0f;   // pico de torque
        float peak_torque_rpm = 4500.0f;
        float inertia_kgm2  = 0.25f;   // inércia do volante
        float friction_coeff= 15.0f;   // atrito interno do motor
    };

    Config cfg;
    float  rpm        = 900.0f;
    float  throttle   = 0.0f;   // 0..1

    // Curva de torque — polinômio que pica no peak_torque_rpm e cai no redline
    float torqueAt(float r) const {
        float t = r / cfg.redline_rpm;   // 0..1 ao longo da faixa

        // Forma de sino: sobe rápido, pica perto de 0.6, desce suave
        float curve = 4.0f*t*(1.0f-t);   // parábola simétrica

        // Ajuste assimétrico: torque máximo um pouco antes do redline
        curve = std::pow(curve, 0.85f);

        return curve * cfg.max_torque_nm;
    }

    // Torque líquido disponível no virabrequim
    float netTorque() const {
        float drive    = torqueAt(rpm) * throttle;
        float friction = cfg.friction_coeff * (rpm / cfg.redline_rpm);
        return drive - friction;
    }

    // Atualiza RPM dado torque de resistência da caixa de câmbio
    void update(float dt, float load_torque) {
        float net  = netTorque() - load_torque;
        float alpha = net / cfg.inertia_kgm2;   // F=Iα → α=F/I
        rpm += alpha * dt * (60.0f / (2.0f * 3.14159f));  // rad/s → RPM
        rpm  = std::clamp(rpm, cfg.idle_rpm, cfg.redline_rpm);
    }
};

class Transmission {
public:
    // Relações de cada marcha (0=neutro, 1..6=marchas)
    float gear_ratios[7] = { 0.0f, 3.85f, 2.16f, 1.45f, 1.10f, 0.88f, 0.74f };
    float final_drive    = 4.1f;
    float efficiency     = 0.87f;
    int   current_gear   = 1;

    float totalRatio() const {
        return gear_ratios[current_gear] * final_drive;
    }

    // Torque nas rodas motrizes
    float wheelTorque(float engine_torque) const {
        return engine_torque * totalRatio() * efficiency;
    }

    // RPM do motor dado a velocidade angular da roda
    float engineRpmFromWheel(float wheel_omega_rads) const {
        float engine_rads = wheel_omega_rads * totalRatio();
        return engine_rads * (60.0f / (2.0f * 3.14159f));
    }

    // Troca de marcha automática simples (heurística)
    void autoShift(float rpm, float throttle) {
        if (rpm > 6800.0f && current_gear < 6) { ++current_gear; }
        if (rpm < 2200.0f && current_gear > 1) { --current_gear; }
    }
};

7. Rodas, Pneus e Suspensão

Suspensão — Modelo Mola-Amortecedor

Cada roda tem uma suspensão independente modelada como um sistema de mola-amortecedor. A força gerada pela suspensão é:

Fsuspensão = −k · x − c · ẋ
// k=rigidez da mola (N/m), x=compressão, c=coeficiente de amortecimento, ẋ=velocidade de compressão
struct SuspensionConfig {
    float rest_length    = 0.35f;   // comprimento de repouso (m)
    float min_length     = 0.10f;   // compressão máxima
    float max_length     = 0.55f;   // extensão máxima
    float spring_rate    = 35000.0f; // N/m
    float damper_rate    = 3500.0f;  // N·s/m
    float anti_roll_bar  = 5000.0f;  // rigidez da barra estabilizadora (N/m)
};

struct SuspensionState {
    float length          = 0.35f;   // comprimento atual
    float velocity        = 0.0f;    // taxa de variação do comprimento
    float force           = 0.0f;    // força resultante (N, positivo=empurra pra cima)
    bool  in_contact      = false;
};

float computeSuspensionForce(const SuspensionConfig& cfg,
                              SuspensionState& state,
                              float contact_length,   // comprimento até o chão
                              float dt)
{
    state.in_contact = (contact_length <= cfg.max_length);
    if (!state.in_contact) {
        state.force = 0.0f;
        return 0.0f;
    }

    float clamped = std::clamp(contact_length, cfg.min_length, cfg.max_length);
    float compression = cfg.rest_length - clamped;  // positivo = comprimida

    // Velocidade de compressão (derivada discreta)
    float new_velocity = (clamped - state.length) / dt;
    state.length   = clamped;
    state.velocity = new_velocity;

    state.force = cfg.spring_rate * compression
                - cfg.damper_rate * new_velocity;

    state.force = std::max(0.0f, state.force);  // suspensão só empurra
    return state.force;
}

Classe Wheel — Estado Completo da Roda

class Wheel {
public:
    struct Config {
        float radius_m       = 0.32f;   // raio do pneu (m)
        float width_m        = 0.25f;
        float inertia_kgm2   = 1.8f;    // inércia rotacional da roda+pneu
        float rolling_resist = 0.015f;  // coeficiente de resistência de rolamento
        bool  is_driven      = false;   // recebe torque do motor?
        bool  is_steered     = false;   // vira com o volante?
        Vec3  attach_local;              // ponto de ancoragem no chassi (local)
        SuspensionConfig suspension;
    };

    Config         cfg;
    SuspensionState susp;

    // Estado dinâmico
    float omega_rads     = 0.0f;  // velocidade angular (rad/s)
    float steer_angle    = 0.0f;  // ângulo de esterçamento (rad)
    float brake_torque   = 0.0f;  // torque de frenagem aplicado
    float drive_torque   = 0.0f;  // torque de tração aplicado
    float normal_force   = 0.0f;  // carga vertical (N)

    // Saídas do modelo de pneu
    Vec3  force_world;            // força total no espaço mundial
    float slip_ratio     = 0.0f;  // escorregamento longitudinal
    float slip_angle_rad = 0.0f;  // ângulo de deslizamento lateral
    bool  is_locked      = false; // roda travada (ABS)?

    // Velocidade linear da superfície do pneu em contato com o chão
    float surfaceSpeed() const { return omega_rads * cfg.radius_m; }

    void updateSpin(float dt) {
        float net_torque = drive_torque
                         - brake_torque
                         - normal_force * cfg.rolling_resist * cfg.radius_m;
        float alpha = net_torque / cfg.inertia_kgm2;
        omega_rads += alpha * dt;
        omega_rads  = std::max(0.0f, omega_rads);  // não gira ao contrário
    }
};

8. Sistema de Frenagem e ABS

A frenagem é modelada como um torque oposto à rotação da roda. O ponto de bloqueio ocorre quando o torque de frenagem supera o atrito disponível no pneu.

Tfreio = Phidráulico · Apistão · μpastilha · rdisco
Ffreio_max = μpneu · Nroda
// Bloqueio quando T_freio > F_freio_max · r_pneu
class BrakeSystem {
public:
    struct Config {
        float max_torque_front = 1800.0f;  // N·m por roda dianteira
        float max_torque_rear  = 1200.0f;  // N·m por roda traseira
        float bias             = 0.6f;     // 60% dianteiro
        bool  abs_enabled      = true;
        float abs_slip_target  = 0.15f;   // slip ratio alvo com ABS ativo
    };

    Config cfg;

    // Calcula torque de frenagem por roda, com ABS digital simples
    float brakeForWheel(float input,          // 0..1
                         float slip_ratio,
                         float normal_force,
                         bool  is_front) const
    {
        float max_T = is_front ? cfg.max_torque_front : cfg.max_torque_rear;
        float bias  = is_front ? cfg.bias : (1.0f - cfg.bias);
        float requested = input * max_T * bias;

        if (cfg.abs_enabled) {
            // ABS: reduz pressão se slip ratio ultrapassou o alvo
            float slip_abs = std::abs(slip_ratio);
            if (slip_abs > cfg.abs_slip_target) {
                float reduction = 1.0f - (slip_abs - cfg.abs_slip_target) * 4.0f;
                requested *= std::clamp(reduction, 0.0f, 1.0f);
            }
        }
        return requested;
    }
};

9. Direção e Esterçamento

O modelo de direção usa a geometria de Ackermann: as rodas dianteiras internas e externas giram ângulos diferentes para manter os centros de curvatura alinhados.

θinner = arctan( L / (R − tw/2) )
θouter = arctan( L / (R + tw/2) )
// L=distância entre eixos, R=raio de curva, tw=bitola
class SteeringSystem {
public:
    struct Config {
        float max_angle_rad    = 0.52f;  // ~30° máximo
        float steering_ratio   = 14.0f;  // volante / roda (14:1)
        float return_speed     = 5.0f;   // velocidade de retorno ao centro
        float speed_sensitivity = 0.3f;  // reduz ângulo em alta velocidade
    };

    Config cfg;
    float  current_angle = 0.0f;

    // Atualiza ângulo das rodas com base no input e velocidade do carro
    void update(float dt, float input, float speed_ms) {
        // Redução progressiva do ângulo em alta velocidade
        float speed_factor = 1.0f / (1.0f + cfg.speed_sensitivity * speed_ms);
        float target = input * cfg.max_angle_rad * speed_factor;

        // Retorno ao centro quando input=0
        if (std::abs(input) < 0.01f) {
            target = lerp(current_angle, 0.0f, cfg.return_speed * dt);
        }
        current_angle = target;
    }

    // Ackermann: ângulo interno e externo para uma dada curvatura
    struct WheelAngles { float left, right; };
    WheelAngles ackermannAngles(float wheelbase, float track) const {
        float a = current_angle;
        if (std::abs(a) < 1e-4f) return {0, 0};

        float R     = wheelbase / std::tan(std::abs(a));
        float inner = std::atan2(wheelbase, R - track * 0.5f);
        float outer = std::atan2(wheelbase, R + track * 0.5f);

        if (a > 0) return { inner,  outer };  // curva à esquerda
        else       return {-outer, -inner };  // curva à direita
    }

private:
    float lerp(float a, float b, float t) { return a + (b-a)*t; }
};

10. Aerodinâmica — Drag e Downforce

O arrasto aerodinâmico cresce com o quadrado da velocidade. Em carros de corrida, o downforce (força para baixo) aumenta a carga nas rodas e melhora o grip em alta velocidade.

Fdrag = ½ · ρ · v² · Cd · A
Fdownforce = ½ · ρ · v² · Cl · A
// ρ=densidade do ar (1.225 kg/m³), Cd=coef. de arrasto, Cl=coef. de sustentação negativa, A=área frontal
class Aerodynamics {
public:
    struct Config {
        float drag_coeff       = 0.32f;   // Cd
        float downforce_coeff  = 0.8f;    // Cl negativo
        float frontal_area_m2  = 2.1f;    // m²
        float air_density      = 1.225f;  // kg/m³ ao nível do mar
        float downforce_bias   = 0.4f;   // 40% dianteiro, 60% traseiro
    };

    Config cfg;

    struct Result {
        Vec3  drag_force;          // oposta à velocidade
        float downforce_front;    // N extra nas rodas dianteiras
        float downforce_rear;     // N extra nas rodas traseiras
    };

    Result compute(const Vec3& velocity_world) const {
        float speed_sq = velocity_world.lengthSq();
        float dyn_p    = 0.5f * cfg.air_density * speed_sq;  // pressão dinâmica

        float drag_mag = dyn_p * cfg.drag_coeff * cfg.frontal_area_m2;
        float df_total = dyn_p * cfg.downforce_coeff * cfg.frontal_area_m2;

        Result r;
        r.drag_force       = velocity_world.normalized() * (-drag_mag);
        r.downforce_front  = df_total * cfg.downforce_bias;
        r.downforce_rear   = df_total * (1.0f - cfg.downforce_bias);
        return r;
    }
};

11. Modelo de Atrito de Pneus (Pacejka)

O modelo de Pacejka (Magic Formula) é a referência da indústria para simular o comportamento de pneus. Ele descreve como a força gerada pelo pneu varia com o slip ratio (longitudinal) e o slip angle (lateral).

F = D · sin( C · arctan( B·x − E·(B·x − arctan(B·x)) ) )
// B=stiffness, C=shape, D=peak, E=curvature — "Magic Formula"

Slip Ratio (Longitudinal)

σ = (vwheel − vcar) / max(vcar, ε)
// σ>0: aceleração (roda girando mais rápido que carro), σ<0: frenagem

Slip Angle (Lateral)

α = arctan( vlateral / vlongitudinal )
// ângulo entre a direção que a roda aponta e a direção que ela se move
struct PacejkaParams {
    float B = 10.0f;   // Stiffness Factor
    float C = 1.9f;    // Shape Factor
    float D = 1.0f;    // Peak Value (multiplicado pela carga normal)
    float E = 0.97f;   // Curvature Factor
};

// Retorna coeficiente de força (0..1), multiplicar pela carga normal
float pacejka(const PacejkaParams& p, float slip) {
    float x = slip;
    return p.D * std::sin(
        p.C * std::atan(
            p.B * x - p.E * (p.B * x - std::atan(p.B * x))
        )
    );
}

class TireModel {
public:
    PacejkaParams long_params  = { 12.0f, 1.65f, 1.0f, 0.97f };
    PacejkaParams lat_params   = { 8.0f,  1.4f,  0.9f, 0.97f };

    struct Input {
        float normal_force;        // N — carga vertical
        float slip_ratio;          // adimensional
        float slip_angle_rad;      // radianos
        float friction_coeff;      // μ da superfície
    };

    struct Output {
        float long_force;    // N — longitudinal (tração/freio)
        float lat_force;     // N — lateral (cornering)
    };

    Output evaluate(const Input& in) const {
        float Fz = in.normal_force;

        // Forças máximas escaladas pela carga e coeficiente de atrito
        float Fx_raw = pacejka(long_params, in.slip_ratio) * Fz * in.friction_coeff;
        float Fy_raw = pacejka(lat_params, in.slip_angle_rad) * Fz * in.friction_coeff;

        // Círculo de fricção: Fx² + Fy² ≤ (μ·Fz)²
        float limit   = in.friction_coeff * Fz;
        float combined = std::sqrt(Fx_raw*Fx_raw + Fy_raw*Fy_raw);
        if (combined > limit) {
            float scale = limit / combined;
            Fx_raw *= scale;
            Fy_raw *= scale;
        }

        return { Fx_raw, Fy_raw };
    }

    // Calcula slip ratio dado velocidade linear do carro e da roda
    static float computeSlipRatio(float wheel_surface_speed,
                                    float car_forward_speed) {
        float denom = std::max(std::abs(car_forward_speed), 0.5f);
        return (wheel_surface_speed - car_forward_speed) / denom;
    }

    // Calcula slip angle dado velocidade local da roda no chassi
    static float computeSlipAngle(float lateral_vel, float longitudinal_vel) {
        if (std::abs(longitudinal_vel) < 0.1f) return 0.0f;
        return std::atan2(lateral_vel, std::abs(longitudinal_vel));
    }
};
Círculo de Fricção
A capacidade total de atrito do pneu é limitada. Usar 100% para frenagem deixa 0% para curva. Este conceito é implementado normalizando o vetor (Fx, Fy) quando seu módulo ultrapassa μ·Fz.

12. Dinâmica Rotacional e Torque

O equivalente rotacional de F=ma é τ = I·α, onde τ é o torque, I é o momento de inércia e α é a aceleração angular.

τ = r × F (produto vetorial da alavanca pelo força)
α = τ / I
// Torque positivo = sentido anti-horário (regra da mão direita)
// Torque do motor aplicado nas rodas motrizes
void applyDriveTorque(CarBody& car, float dt) {
    float engine_torque = car.engine.netTorque();
    float wheel_torque  = car.transmission.wheelTorque(engine_torque);

    // Distribuição simples: 50/50 entre rodas motrizes traseiras
    float per_wheel = wheel_torque * 0.5f;
    car.wheels[static_cast<int>(WheelIndex::RearLeft)].drive_torque  = per_wheel;
    car.wheels[static_cast<int>(WheelIndex::RearRight)].drive_torque = per_wheel;
}

// Torque de yaw gerado por força lateral desigual entre eixos
float computeYawTorque(const std::array<Wheel, 4>& wheels,
                        float wheelbase)
{
    // Momento gerado pela diferença lateral entre eixo dianteiro e traseiro
    float Fy_front = wheels[0].force_world.x + wheels[1].force_world.x;
    float Fy_rear  = wheels[2].force_world.x + wheels[3].force_world.x;

    float a = wheelbase * 0.45f;  // distância CG → eixo dianteiro
    float b = wheelbase * 0.55f;  // distância CG → eixo traseiro

    return Fy_front * a - Fy_rear * b;
}

13. Colisões e Impulsos

Quando o carro colide, usamos impulsos para alterar instantaneamente a velocidade do corpo. O impulso J é calculado a partir do coeficiente de restituição e das massas envolvidas:

J = −(1 + e) · vrel · n̂ / (1/mA + 1/mB)
// e=coeficiente de restituição (0=perfeitamente inelástico, 1=elástico)
struct ContactManifold {
    Vec3  point;         // ponto de contato no mundo
    Vec3  normal;        // normal apontando de B → A
    float penetration;   // profundidade de interpenetração (m)
};

void resolveCollision(RigidBody3D& A, RigidBody3D& B,
                       const ContactManifold& c,
                       float restitution = 0.3f,
                       float friction_mu = 0.5f)
{
    Vec3 rA = c.point - A.position;
    Vec3 rB = c.point - B.position;

    // Velocidade relativa no ponto de contato
    Vec3 vA = A.velocity + A.angularVelocity.cross(rA);
    Vec3 vB = B.velocity + B.angularVelocity.cross(rB);
    float vRel_n = (vA - vB).dot(c.normal);

    if (vRel_n >= 0) return;  // objetos já se afastando

    // Denominador do impulso (considera inércia rotacional)
    auto invInertia = [](RigidBody3D& rb, const Vec3& r, const Vec3& n) {
        Vec3 cross = r.cross(n);
        return Vec3{
            cross.x / rb.inertiaTensor.x,
            cross.y / rb.inertiaTensor.y,
            cross.z / rb.inertiaTensor.z
        }.cross(r).dot(n);
    };

    float invMassSum = 1.0f/A.mass + 1.0f/B.mass
                     + invInertia(A, rA, c.normal)
                     + invInertia(B, rB, c.normal);

    float j = -((1.0f + restitution) * vRel_n) / invMassSum;

    Vec3 impulse = c.normal * j;
    A.velocity += impulse * (1.0f / A.mass);
    B.velocity -= impulse * (1.0f / B.mass);
    A.angularVelocity += Vec3{
        rA.cross(impulse).x / A.inertiaTensor.x,
        rA.cross(impulse).y / A.inertiaTensor.y,
        rA.cross(impulse).z / A.inertiaTensor.z
    };

    // Correção de posição (evita afundamento)
    float slop    = 0.01f;
    float percent  = 0.4f;
    Vec3  correction = c.normal *
        (std::max(c.penetration - slop, 0.0f) / invMassSum * percent);
    A.position += correction * (1.0f/A.mass);
    B.position -= correction * (1.0f/B.mass);
}

14. Loop de Integração Física

O sistema de física roda em um timestep fixo separado do framerate de renderização. O padrão recomendado é 120 Hz (dt = 1/120 ≈ 8.33ms).

class PhysicsWorld {
public:
    static constexpr float FIXED_DT   = 1.0f / 120.0f;
    static constexpr float MAX_STEPS  = 8;     // evita espiral da morte
    static constexpr float GRAVITY    = 9.81f;

    void stepFrame(float real_dt, CarBody& car,
                   float throttle, float brake, float steer)
    {
        accumulator += real_dt;
        int steps = 0;

        while (accumulator >= FIXED_DT && steps < MAX_STEPS) {
            stepFixed(FIXED_DT, car, throttle, brake, steer);
            accumulator -= FIXED_DT;
            ++steps;
        }
        // alpha = accumulator/FIXED_DT → usar para interpolação de renderização
    }

private:
    float accumulator = 0.0f;

    void stepFixed(float dt, CarBody& car,
                    float throttle, float brake, float steer)
    {
        // 1. Atualizar inputs
        car.input_throttle = throttle;
        car.input_brake    = brake;
        car.input_steer    = steer;

        // 2. Atualizar direção
        car.steering.update(dt, steer, car.body.speedKmh() / 3.6f);

        // 3. Calcular carga por roda (transferência de peso)
        Vec3 accel_local = car.body.orientation.rotate(car.body.acceleration);
        WheelLoad loads = computeWheelLoads(car._cfg, accel_local);

        // 4. Processar cada roda
        for (int i = 0; i < 4; ++i) {
            Wheel& w = car.wheels[i];
            float Fz = (&loads.fl)[i];

            // Raycast suspensão para detectar chão
            float contact_len = rayCastGround(car.body, w.cfg.attach_local);
            float Fsusp = computeSuspensionForce(w.cfg.suspension, w.susp, contact_len, dt);

            w.normal_force = Fsusp + Fz;

            // Velocidade da roda no espaço local
            Vec3 wp_world  = car.body.orientation.rotate(w.cfg.attach_local) + car.body.position;
            Vec3 vel_world = car.body.velocity
                           + car.body.angularVelocity.cross(wp_world - car.body.position);
            Vec3 vel_local = car.body.orientation.rotate(vel_world);  // simplificado

            // Slip ratio e ângulo
            w.slip_ratio    = TireModel::computeSlipRatio(w.surfaceSpeed(), vel_local.z);
            w.slip_angle_rad= TireModel::computeSlipAngle(vel_local.x, vel_local.z);

            // Forças do pneu
            TireModel::Input tin{ w.normal_force, w.slip_ratio, w.slip_angle_rad, 0.85f };
            TireModel::Output tout = car.tire_model.evaluate(tin);

            // Converte para espaço mundial e aplica no corpo
            Vec3 wheel_fwd = car.body.orientation.rotate({0,0,1});
            Vec3 wheel_right= car.body.orientation.rotate({1,0,0});
            Vec3 tire_force = wheel_fwd * tout.long_force + wheel_right * tout.lat_force;
            Vec3 susp_force = car.body.up() * Fsusp;

            car.body.addForceAtPoint(tire_force + susp_force, wp_world);

            // Atualiza spin da roda
            w.drive_torque = /* calculado pelo motor */ 0.0f;
            w.brake_torque = car.brake_system.brakeForWheel(
                brake, w.slip_ratio, w.normal_force, i < 2);
            w.updateSpin(dt);
        }

        // 5. Gravidade
        car.body.addForce({0, -GRAVITY * car._cfg.mass_kg, 0});

        // 6. Aerodinâmica
        auto aero = car.aero.compute(car.body.velocity);
        car.body.addForce(aero.drag_force);

        // 7. Motor e câmbio
        applyDriveTorque(car, dt);
        car.engine.update(dt, 0.0f);
        car.transmission.autoShift(car.engine.rpm, throttle);

        // 8. Integrar corpo rígido
        car.body.integrate(dt);
    }
};
